5. application.properties 파일에서 다음 설정 확인
	- src/main/resources/application.properties

	# 1. 애플리케이션 정보 및 네트워크 설정
	# 애플리케이션의 이름을 정의. 서비스 식별 및 모니터링 시 사용.
	spring.application.name=web1
	# 서버가 실행될 포트 번호. 브라우저에서 http://localhost:9991 로 접속.
	server.port=9991

	# 애플리케이션의 기본 경로(Context Path)를 설정. '/'는 루트 경로를 의미.
	server.servlet.context-path=/

	# 2. 로깅 설정
	# 전체 로그 수준을 INFO(정보성)로 설정.
	logging.level.root=info
	# 특정 패키지(net.datasa.web1)만 DEBUG 수준으로 로그를 남겨 개발 시 상세 추적이 가능하게 설정.
	logging.level.net.datasa.web1=debug

7. Controller 생성
	- src/main/java/패키지명을 마우스 우클릭 - New - Java Class - 클래스명 (HomeController) 입력
----------------------------------------------------------------------------------------------------------------------------------------
	package net.datasa.web1;

	import org.springframework.stereotype.Controller;
	import org.springframework.web.bind.annotation.GetMapping;

	// @Controller: 이 클래스가 웹 요청을 처리하는 컨트롤러임을 Spring에 알림
	@Controller
	public class HomeController {

		// @GetMapping: 특정 URL(root)로 들어오는 요청을 이 메서드에서 처리
		@GetMapping({"", "/"})
		public String home() {

			System.out.println("Home Controller 실행");

			// "home"을 반환하면 templates/home.html 파일을 찾아 화면에 표시
			return "home";
		}
	}
----------------------------------------------------------------------------------------------------------------------------------------

	- 웹브라우저에서 http://localhost:9991 접속
